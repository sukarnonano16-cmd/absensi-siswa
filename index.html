ℹ️ Informasi Sistem

Versi: SIAS 1.0 • Penyimpanan: browser localStorage (offline) + Sinkron Cloud opsional via Google Drive (Apps Script) (multi perangkat, saat online). Atur di menu Sinkron Cloud.

Sinkron Cloud (Multi Perangkat)

Hubungkan aplikasi ke Google Drive milikmu lewat Google Apps Script (gratis selamanya, tanpa kartu kredit) agar data siswa, kelas, dan absensi otomatis tersinkron ke semua laptop & HP yang memakai URL & Kode Rahasia yang sama. Jika offline, data tetap tersimpan dan dikirim otomatis saat online.

Status: Belum terhubung

Cara Menyiapkan (Gratis, ±5 menit, cukup sekali)

  1. Buka script.google.com, login akun Google, klik Proyek baru (New project).
  2. Hapus semua isi Code.gs, lalu tempel kode di bawah (tombol Salin Kode). Ganti tulisan GANTI-DENGAN-KODE-RAHASIA-ANDA pada baris KUNCI dengan kode rahasia buatanmu sendiri (panjang & acak), lalu klik ikon simpan.
  3. Klik Deploy (Terapkan) → Deployment baru, pilih jenis Aplikasi web (Web app). Isi Execute as: Me (Saya) dan Who has access: Anyone (Siapa saja), lalu Deploy. Saat diminta izin, pilih akunmu → Advanced → Go to ... (unsafe) → Allow (peringatan ini normal untuk skrip buatan sendiri).
  4. Salin Web app URL (berakhiran /exec), tempel di kolom URL di atas. Isi Kode Rahasia sama persis dengan KUNCI, lalu klik Hubungkan & Sinkron.
  5. Di laptop/HP lain, buka aplikasi ini, lalu isi URL dan Kode Rahasia yang sama persis.
// SIAS - Server sinkron v3 (Google Apps Script) - cepat: cache + tanpa baca Drive saat cek perubahan
const KUNCI = 'GANTI-DENGAN-KODE-RAHASIA-ANDA';
const NAMA_FILE = 'SIAS_DATA.json';
const MAKS_LOG = 300000; // batas catatan perubahan (karakter)
const C = CacheService.getScriptCache(), P = PropertiesService.getScriptProperties();

function doPost(e) {
  let req;
  try { req = JSON.parse(e.postData.contents); } catch (x) { return out({ ok: false, error: 'permintaan rusak' }); }
  if (req.code !== KUNCI) return out({ ok: false, error: 'kunci salah' });
  const tulisAksi = req.action === 'put' || req.action === 'patch';
  const lock = LockService.getScriptLock();
  try {
    if (tulisAksi) lock.waitLock(25000); // cek/baca tidak perlu kunci = lebih cepat
    const rev = ambilRev(); // null = data belum ada
    const ada = rev !== null, r0 = rev || 0;
    if (req.action === 'meta') return out({ ok: true, exists: ada, updatedAt: r0 });
    if (req.action === 'get') {
      if (!ada) return out({ ok: true, exists: false, updatedAt: 0 });
      return out({ ok: true, exists: true, updatedAt: r0, payload: JSON.stringify(muat(r0).db) });
    }
    if (req.action === 'put') {
      if (ada && r0 !== Number(req.expect || 0)) return out({ ok: false, error: 'bentrok' });
      const t = Number(req.updatedAt);
      simpan({ v: 2, db: JSON.parse(req.payload), log: [], logBase: t }, t);
      return out({ ok: true, updatedAt: t });
    }
    if (req.action === 'pull') {
      if (!ada) return out({ ok: true, exists: false, updatedAt: 0 });
      const since = Number(req.since || 0);
      if (since === r0) return out({ ok: true, exists: true, same: true, updatedAt: r0 });
      const w = muat(r0);
      if (since > r0 || since < w.logBase) return out({ ok: true, exists: true, full: true, updatedAt: r0, payload: JSON.stringify(w.db) });
      return out({ ok: true, exists: true, updatedAt: r0, ops: opsSejak(w, since) });
    }
    if (req.action === 'patch') {
      if (!ada) return out({ ok: false, error: 'data belum ada' });
      const ops = req.ops || [], since = Number(req.since || 0);
      if (!ops.length && since === r0) return out({ ok: true, exists: true, same: true, updatedAt: r0 });
      const w = muat(r0);
      const penuh = since > r0 || since < w.logBase;
      const missed = penuh ? null : opsSejak(w, since);
      if (!ops.length) return penuh ? out({ ok: true, full: true, updatedAt: r0, payload: JSON.stringify(w.db) }) : out({ ok: true, updatedAt: r0, ops: missed });
      const baru = Math.max(Date.now(), r0 + 1);
      terapkan(w.db, ops);
      w.log.push({ r: baru, o: ops });
      let tot = JSON.stringify(w.log).length;
      while (w.log.length > 1 && tot > MAKS_LOG) { const b = w.log.shift(); tot -= JSON.stringify(b).length + 1; w.logBase = b.r; }
      simpan(w, baru);
      return penuh ? out({ ok: true, full: true, updatedAt: baru, payload: JSON.stringify(w.db) }) : out({ ok: true, updatedAt: baru, ops: missed });
    }
    return out({ ok: false, error: 'aksi tidak dikenal' });
  } catch (err) {
    return out({ ok: false, error: String(err) });
  } finally {
    try { lock.releaseLock(); } catch (x) {}
  }
}
function doGet() { return out({ ok: true, info: 'SIAS sinkron aktif (v3)' }); }

function kunci(x) { return (x && x.id) || JSON.stringify(x); }
function terapkan(db, ops) {
  const peta = {}, sentuh = {};
  const idx = (col) => {
    if (!peta[col]) { peta[col] = new Map(); (db[col] || []).forEach((x, i) => peta[col].set(kunci(x), i)); }
    return peta[col];
  };
  ops.forEach(op => {
    const k = op[0], v = op[1], t = k.charAt(0), r = k.slice(2);
    if (t === 'a') {
      const i = r.indexOf('|'), d = r.slice(0, i), s = r.slice(i + 1);
      db.attendance = db.attendance || {};
      if (v === null) { if (db.attendance[d]) { delete db.attendance[d][s]; if (!Object.keys(db.attendance[d]).length) delete db.attendance[d]; } }
      else { (db.attendance[d] = db.attendance[d] || {})[s] = v; }
    } else if (t === 'u') {
      const i = r.indexOf('|'), col = r.slice(0, i), id = r.slice(i + 1);
      db[col] = db[col] || [];
      const m = idx(col); sentuh[col] = 1;
      if (v === null) { if (m.has(id)) { db[col][m.get(id)] = null; m.delete(id); } }
      else { const o = JSON.parse(v); if (m.has(id)) db[col][m.get(id)] = o; else { db[col].push(o); m.set(id, db[col].length - 1); } }
    } else if (t === 'k') {
      if (v === null) delete db[r]; else db[r] = JSON.parse(v);
    }
  });
  Object.keys(sentuh).forEach(c => { db[c] = db[c].filter(x => x !== null); });
}
function opsSejak(w, since) {
  let hasil = [];
  w.log.forEach(b => { if (b.r > since) hasil = hasil.concat(b.o); });
  return hasil;
}
// ---- penyimpanan: Drive (permanen) + Cache (cepat, dikompres) ----
function cariFile() {
  const id = P.getProperty('FID');
  if (id) { try { return DriveApp.getFileById(id); } catch (x) {} }
  const it = DriveApp.getFilesByName(NAMA_FILE);
  if (!it.hasNext()) return null;
  const f = it.next(); P.setProperty('FID', f.getId()); return f;
}
function ambilRev() {
  const c = C.get('rev');
  if (c !== null) return Number(c);
  const f = cariFile();
  if (!f) return null;
  const r = Number(f.getDescription() || 0);
  C.put('rev', String(r), 21600);
  return r;
}
function muat(rev) {
  let w = bacaCache(rev);
  if (w) return w;
  const f = cariFile();
  const j = JSON.parse(f.getBlob().getDataAsString());
  w = (j && j.v === 2 && j.db) ? { v: 2, db: j.db, log: j.log || [], logBase: Number(j.logBase || 0) } : { v: 2, db: j, log: [], logBase: rev };
  cacheTulis(w, rev);
  return w;
}
function simpan(w, rev) {
  const isi = JSON.stringify(w);
  const f = cariFile();
  if (f) { f.setContent(isi); f.setDescription(String(rev)); }
  else { const n = DriveApp.createFile(NAMA_FILE, isi, MimeType.PLAIN_TEXT); n.setDescription(String(rev)); P.setProperty('FID', n.getId()); }
  cacheTulis(w, rev, isi);
}
function cacheTulis(w, rev, isi) {
  try {
    const z = Utilities.base64Encode(Utilities.gzip(Utilities.newBlob(isi || JSON.stringify(w), 'text/plain', 'd.txt')).getBytes());
    const o = {}, n = Math.ceil(z.length / 90000);
    for (let i = 0; i < n; i++) o['s' + i] = z.substr(i * 90000, 90000);
    o.sn = String(n); o.sr = String(rev); o.rev = String(rev);
    C.putAll(o, 21600);
  } catch (x) { C.remove('rev'); } // gagal cache -> berikutnya baca dari Drive
}
function bacaCache(rev) {
  try {
    if (C.get('sr') !== String(rev)) return null;
    const n = Number(C.get('sn') || 0);
    if (!n) return null;
    const ks = []; for (let i = 0; i < n; i++) ks.push('s' + i);
    const m = C.getAll(ks); let z = '';
    for (let i = 0; i < n; i++) { if (!m['s' + i]) return null; z += m['s' + i]; }
    return JSON.parse(Utilities.ungzip(Utilities.newBlob(Utilities.base64Decode(z), 'application/x-gzip', 'd.gz')).getDataAsString());
  } catch (x) { return null; }
}
function out(o) {
  return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON);
}

Catatan: data tersimpan sebagai file SIAS_DATA.json di Google Drive-mu sendiri. Sudah punya script lama? Ganti seluruh isi Code.gs dengan kode ini (isi KUNCI sama persis dengan yang lama), lalu Deploy → Kelola deployment → Edit → Versi baru → Deploy. URL dan kode rahasia tidak berubah, data lama tetap aman. Kode v2 hanya mengirim perubahan sehingga jauh lebih cepat; aplikasi otomatis memakai mode lama bila script belum diperbarui. Perangkat lain memeriksa pembaruan tiap ±6 detik saat aplikasi terbuka.
